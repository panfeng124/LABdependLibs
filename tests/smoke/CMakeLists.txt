cmake_minimum_required(VERSION 3.22)
project(LABDepsSmoke LANGUAGES C CXX)
find_package(LABDeps CONFIG REQUIRED)
add_executable(lab-deps-smoke smoke.cpp)
target_compile_features(lab-deps-smoke PRIVATE cxx_std_20)
target_link_libraries(lab-deps-smoke PRIVATE LAB::all)
set_property(TARGET lab-deps-smoke PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
foreach(name IN ITEMS zlib brotli libpng freetype harfbuzz glfw glm libhv lunasvg nfd openal yoga assimp sqlite3 libffi)
    if(TARGET LAB::${name})
        string(TOUPPER "${name}" upper)
        target_compile_definitions(lab-deps-smoke PRIVATE "LAB_HAVE_${upper}=1")
    endif()
endforeach()
lab_deps_copy_runtime(lab-deps-smoke)
enable_testing()
add_test(NAME consume-packaged-libraries COMMAND lab-deps-smoke)
